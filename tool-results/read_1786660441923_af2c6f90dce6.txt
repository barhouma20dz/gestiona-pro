    85→/**
    86→ * Mini sparkline — renders an inline SVG polyline of the last 7 days' revenue.
    87→ * Each point's Y is proportional to the max value; zero points sit on the baseline.
    88→ */
    89→function RevenueSparkline({
    90→  values,
    91→  className,
    92→}: {
    93→  values: number[];
    94→  className?: string;
    95→}) {
    96→  const width = 120;
    97→  const height = 36;
    98→  const pad = 3;
    99→  const max = Math.max(...values, 1); // avoid divide-by-zero
   100→  const innerH = height - pad * 2;
   101→  const step = values.length > 1 ? (width - pad * 2) / (values.length - 1) : 0;
   102→
   103→  const points = values.map((v, i) => {
   104→    const x = pad + i * step;
   105→    const y = pad + innerH - (v / max) * innerH;
   106→    return `${x.toFixed(1)},${y.toFixed(1)}`;
   107→  });
   108→  const path = points.join(" ");
   109→
   110→  // Area path (fill under the line)
   111→  const areaPath =
   112→    values.length > 0
   113→      ? `M ${pad},${(pad + innerH).toFixed(1)} L ${path
   114→          .replace(/ /g, " L ")} L ${(pad + (values.length - 1) * step).toFixed(
   115→            1
   116→          )},${(pad + innerH).toFixed(1)} Z`
   117→      : "";
   118→
   119→  return (
   120→    <svg
   121→      width={width}
   122→      height={height}
   123→      viewBox={`0 0 ${width} ${height}`}
   124→      className={cn("overflow-visible", className)}
   125→      preserveAspectRatio="none"
   126→    >
   127→      <defs>
   128→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   129→          <stop offset="0%" stopColor="currentColor" stopOpacity="0.3" />
   130→          <stop offset="100%" stopColor="currentColor" stopOpacity="0" />
   131→        </linearGradient>
   132→      </defs>
   133→      {areaPath && <path d={areaPath} fill="url(#sparkGrad)" />}
   134→      <polyline
   135→        points={path}
   136→        fill="none"
   137→        stroke="currentColor"
   138→        strokeWidth="1.75"
   139→        strokeLinecap="round"
   140→        strokeLinejoin="round"
   141→      />
   142→      {values.map((v, i) => {
   143→        const x = pad + i * step;
   144→        const y = pad + innerH - (v / max) * innerH;
   145→        const isLast = i === values.length - 1;
   146→        return (
   147→          <circle
   148→            key={i}
   149→            cx={x}
   150→            cy={y}
   151→            r={isLast ? 2.5 : 0}
   152→            fill="currentColor"
   153→          />
   154→        );
   155→      })}
   156→    </svg>
   157→  );
   158→}

   233→/**
   234→ * Donut chart — renders an SVG conic-gradient-like donut for payment breakdown.
   235→ * Each segment is an SVG stroke-dasharray on a circle.
   236→ */
   237→function PaymentDonut({
   238→  segments,
   239→}: {
   240→  segments: { method: string; percent: number }[];
   241→}) {
   242→  const size = 128;
   243→  const stroke = 16;
   244→  const radius = (size - stroke) / 2;
   245→  const circumference = 2 * Math.PI * radius;
   246→
   247→  // Precompute cumulative offsets so we don't mutate during render.
   248→  const segs = segments.reduce<
   249→    { method: string; percent: number; dash: number; offset: number }[]
   250→  >((acc, seg) => {
   251→    const dash = (seg.percent / 100) * circumference;
   252→    const offset = acc.length > 0 ? acc[acc.length - 1].offset + acc[acc.length - 1].dash : 0;
   253→    acc.push({ method: seg.method, percent: seg.percent, dash, offset });
   254→    return acc;
   255→  }, []);
   256→
   257→  return (
   258→    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   259→      {/* Background ring */}
   260→      <circle
   261→        cx={size / 2}
   262→        cy={size / 2}
   263→        r={radius}
   264→        fill="none"
   265→        stroke="currentColor"
   266→        strokeWidth={stroke}
   267→        className="text-muted"
   268→      />
   269→      {segs.map((seg, i) => (
   270→        <circle
   271→          key={`${seg.method}-${i}`}
   272→          cx={size / 2}
   273→          cy={size / 2}
   274→          r={radius}
   275→          fill="none"
   276→          stroke={paymentColor(seg.method)}
   277→          strokeWidth={stroke}
   278→          strokeDasharray={`${seg.dash} ${circumference - seg.dash}`}
   279→          strokeDashoffset={-seg.offset}
   280→          transform={`rotate(-90 ${size / 2} ${size / 2})`}
   281→        />
   282→      ))}
   283→      {/* Center label */}
   284→      <text
   285→        x="50%"
   286→        y="50%"
   287→        textAnchor="middle"
   288→        dominantBaseline="central"
   289→        className="fill-foreground text-lg font-bold"
   290→      >
   291→        {segments.length}
   292→      </text>
   293→      <text
   294→        x="50%"
   295→        y="50%"
   296→        dy="16"
   297→        textAnchor="middle"
   298→        dominantBaseline="central"
   299→        className="fill-muted-foreground text-[9px] font-medium uppercase tracking-wider"
   300→      >
   301→        methods
   302→      </text>
   303→    </svg>
   304→  );
   305→}

   346→export function Dashboard({
   347→  onNavigate,
   348→}: {
   349→  onNavigate: (view: ViewKey) => void;
   350→}) {
   351→  /* queries */
   352→  const shopQuery = useQuery({ queryKey: ["shop"], queryFn: shopApi.get });
   353→  const statsQuery = useQuery({
   354→    queryKey: ["salesStats"],
   355→    queryFn: salesApi.stats,
   356→  });
   357→  const productsQuery = useQuery({
   358→    queryKey: ["products"],
   359→    queryFn: productsApi.list,
   360→  });
   361→  const salesQuery = useQuery({ queryKey: ["sales"], queryFn: salesApi.list });
   362→
   363→  const shop = shopQuery.data;
   364→  const currency = shop?.currency ?? "₹";
   365→  const stats = statsQuery.data;
   366→  const products = productsQuery.data ?? [];
   367→  const sales = salesQuery.data ?? [];
   368→
   369→  /* derived */
   370→  const inventoryValue = useMemo(
   371→    () => products.reduce((sum, p) => sum + p.price * p.stock, 0),
   372→    [products]
   373→  );
   374→  const lowStockCount = useMemo(
   375→    () => products.filter((p) => p.stock >= 1 && p.stock <= 5).length,
   376→    [products]
   377→  );
   378→  const outOfStockCount = useMemo(
   379→    () => products.filter((p) => p.stock === 0).length,
   380→    [products]
   381→  );
   382→  const healthyCount = Math.max(
   383→    0,
   384→    products.length - lowStockCount - outOfStockCount
   385→  );
   386→
   387→  const topLowStock = useMemo(
   388→    () => [...products].sort((a, b) => a.stock - b.stock).slice(0, 5),
   389→    [products]
   390→  );
   391→  const recentSales = useMemo(() => sales.slice(0, 5), [sales]);
   392→
   393→  // 7-day revenue trend for the sparkline (oldest → newest, today is last)
   394→  const revenueTrend = useMemo(() => {
   395→    const days: number[] = [];
   396→    const now = new Date();
   397→    for (let i = 6; i >= 0; i--) {
   398→      const day = new Date(now);
   399→      day.setHours(0, 0, 0, 0);
   400→      day.setDate(day.getDate() - i);
   401→      const next = new Date(day);
   402→      next.setDate(next.getDate() + 1);
   403→      const total = sales
   404→        .filter((s) => {
   405→          const d = new Date(s.createdAt);
   406→          return d >= day && d < next;
   407→        })
   408→        .reduce((sum, s) => sum + s.total, 0);
   409→      days.push(total);
   410→    }
   411→    return days;
   412→  }, [sales]);
   413→  const weekTotal = revenueTrend.reduce((a, b) => a + b, 0);
   414→  const prevDaysTotal = revenueTrend.slice(0, 6).reduce((a, b) => a + b, 0);
   415→  const todayVsPrev =
   416→    prevDaysTotal === 0
   417→      ? null
   418→      : ((revenueTrend[6] - prevDaysTotal / 6) / (prevDaysTotal / 6)) * 100;
   419→
   420→  // Top selling products (by quantity sold across all sales)
   421→  const topProducts = useMemo(() => {
   422→    const map = new Map<
   423→      string,
   424→      { name: string; qty: number; revenue: number }
   425→    >();
   426→    sales.forEach((sale) => {
   427→      sale.items.forEach((item) => {
   428→        const existing = map.get(item.productName) ?? {
   429→          name: item.productName,
   430→          qty: 0,
   431→          revenue: 0,
   432→        };
   433→        existing.qty += item.quantity;
   434→        existing.revenue += item.total;
   435→        map.set(item.productName, existing);
   436→      });
   437→    });
   438→    return Array.from(map.values())
   439→      .sort((a, b) => b.qty - a.qty)
   440→      .slice(0, 5);
   441→  }, [sales]);
   442→
   443→  // Payment method breakdown (by revenue)
   444→  const paymentBreakdown = useMemo(() => {
   445→    const map = new Map<string, { count: number; revenue: number }>();
   446→    sales.forEach((sale) => {
   447→      const key = sale.paymentMethod || "Cash";
   448→      const existing = map.get(key) ?? { count: 0, revenue: 0 };
   449→      existing.count += 1;
   450→      existing.revenue += sale.total;
   451→      map.set(key, existing);
   452→    });
   453→    const total = sales.length;
   454→    return Array.from(map.entries()).map(([method, data]) => ({
   455→      method,
   456→      ...data,
   457→      percent: total > 0 ? (data.count / total) * 100 : 0,
   458→    }));
   459→  }, [sales]);
   460→
   461→  // Sales by hour (last 7 days, 24 buckets) for the heatmap
   462→  const hourlyBuckets = useMemo(() => {
   463→    const buckets = new Array(24).fill(0) as number[];
   464→    const cutoff = new Date();
   465→    cutoff.setDate(cutoff.getDate() - 7);
   466→    sales.forEach((sale) => {
   467→      const d = new Date(sale.createdAt);
   468→      if (d >= cutoff) {
   469→        buckets[d.getHours()] += sale.total;
   470→      }
   471→    });
   472→    return buckets;
   473→  }, [sales]);
   474→  const maxHourly = Math.max(1, ...hourlyBuckets);
   475→  const peakHour = hourlyBuckets.indexOf(Math.max(...hourlyBuckets));
   476→  const totalHourly = hourlyBuckets.reduce((a, b) => a + b, 0);
   477→
   478→  // Inventory turnover ratio: units sold (7d) / avg units in stock
   479→  const inventoryTurnover = useMemo(() => {
   480→    const cutoff = new Date();
   481→    cutoff.setDate(cutoff.getDate() - 7);
   482→    let unitsSold7d = 0;
   483→    sales.forEach((sale) => {
   484→      if (new Date(sale.createdAt) >= cutoff) {
   485→        sale.items.forEach((it) => {
   486→          unitsSold7d += it.quantity;
   487→        });
   488→      }
   489→    });
   490→    const avgStock =
   491→      products.length > 0
   492→        ? products.reduce((sum, p) => sum + p.stock, 0) / products.length
   493→        : 0;
   494→    const turnover = avgStock > 0 ? unitsSold7d / avgStock : 0;
   495→    // Turnover rate: low (<0.3 = slow), medium (0.3-1), high (>1)
   496→    const rate =
   497→      turnover === 0
   498→        ? "none"
   499→        : turnover < 0.3
   500→        ? "slow"
   501→        : turnover < 1
   502→        ? "medium"
   503→        : "high";
   504→    return { unitsSold7d, avgStock, turnover, rate };
   505→  }, [sales, products]);
   506→
   507→  // Best-selling day of week (from all sales)
   508→  const dayOfWeekStats = useMemo(() => {
   509→    const days = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
   510→    const buckets = new Array(7).fill(0) as number[];
   511→    let totalRevenue = 0;
   512→    sales.forEach((sale) => {
   513→      const dow = new Date(sale.createdAt).getDay();
   514→      buckets[dow] += sale.total;
   515→      totalRevenue += sale.total;
   516→    });
   517→    const maxIdx = buckets.indexOf(Math.max(...buckets));
   518→    return {
   519→      days,
   520→      buckets,
   521→      bestDay: days[maxIdx],
   522→      bestDayRevenue: buckets[maxIdx],
   523→      totalRevenue,
   524→    };
   525→  }, [sales]);
   526→
   527→  /* greeting */
   528→  const now = new Date();
   529→  const fullDate = now.toLocaleDateString("en-US", {
   530→    weekday: "long",
   531→    day: "numeric",
   532→    month: "long",
   533→    year: "numeric",
   534→  });
   535→
   536→  /* KPI cards */
   537→  const kpis: {
   538→    label: string;
   539→    value: string;
   540→    numericValue: number;
   541→    decimals: number;
   542→    icon: typeof TrendingUp;
   543→    tint: string;
   544→    subtitle: string;
   545→    comparison?: boolean;
   546→    clickable?: boolean;
   547→    loading: boolean;
   548→  }[] = [
   549→    {
   550→      label: "Today's Revenue",
   551→      value: formatMoney(stats?.totalSales ?? 0, currency),
   552→      numericValue: stats?.totalSales ?? 0,
   553→      decimals: 2,
   554→      icon: TrendingUp,
   555→      tint: "bg-emerald-500/15 text-emerald-600",
   556→      subtitle: `from ${stats?.transactionCount ?? 0} sales`,
   557→      comparison: true,
   558→      loading: statsQuery.isLoading,
   559→    },
   560→    {
   561→      label: "Inventory Value",
   562→      value: formatMoney(inventoryValue, currency),
   563→      numericValue: inventoryValue,
   564→      decimals: 2,
   565→      icon: Boxes,
   566→      tint: "bg-amber-500/15 text-amber-600",
   567→      subtitle: `${products.length} products`,
   568→      loading: productsQuery.isLoading,
   569→    },
   570→    {
   571→      label: "Low Stock Alerts",
   572→      value: String(lowStockCount),
   573→      numericValue: lowStockCount,
   574→      decimals: 0,
   575→      icon: AlertTriangle,
   576→      tint: "bg-orange-500/15 text-orange-600",
   577→      subtitle: `${outOfStockCount} out of stock`,
   578→      clickable: true,
   579→      loading: productsQuery.isLoading,
   580→    },
   581→    {
   582→      label: "Avg Ticket",
   583→      value: formatMoney(stats?.avgTicket ?? 0, currency),
   584→      numericValue: stats?.avgTicket ?? 0,
   585→      decimals: 2,
   586→      icon: Receipt,
   587→      tint: "bg-slate-500/15 text-slate-600",
   588→      subtitle: "per transaction today",
   589→      loading: statsQuery.isLoading,
   590→    },
   591→  ];
   592→
   593→  return (
   594→    <div className="space-y-6">
   595→      {/* 1. Welcome header — gradient hero */}
   596→      <div className="relative overflow-hidden rounded-2xl border bg-gradient-to-br from-emerald-500 via-emerald-600 to-teal-700 p-6 text-white shadow-lg sm:p-8">
   597→        {/* Decorative circles */}
   598→        <div className="pointer-events-none absolute -right-16 -top-16 h-48 w-48 rounded-full bg-white/10 blur-2xl" />
   599→        <div className="pointer-events-none absolute -bottom-20 -left-10 h-40 w-40 rounded-full bg-white/5 blur-2xl" />
   600→        <div className="relative flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   601→          <div className="min-w-0">
   602→            <p className="text-sm font-medium text-emerald-50/90">
   603→              Welcome back
   604→            </p>
   605→            <h1 className="mt-1 text-2xl font-bold tracking-tight drop-shadow-sm sm:text-3xl">
   606→              {shopQuery.isLoading ? (
   607→                <Skeleton className="h-8 w-48 bg-white/20" />
   608→              ) : (
   609→                shop?.name ?? "SwiftBill POS"
