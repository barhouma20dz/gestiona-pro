    75→/**
    76→ * Build a 7-day revenue series (oldest → newest) from a sales list.
    77→ * Each entry: { label, key, total }.
    78→ */
    79→function build7DaySeries(sales: Sale[]) {
    80→  const days: { date: Date; label: string; total: number }[] = [];
    81→  const today = new Date();
    82→  today.setHours(0, 0, 0, 0);
    83→
    84→  for (let i = 6; i >= 0; i--) {
    85→    const d = new Date(today);
    86→    d.setDate(today.getDate() - i);
    87→    days.push({
    88→      date: d,
    89→      label: d.toLocaleDateString("en-US", { weekday: "short" }),
    90→      total: 0,
    91→    });
    92→  }
    93→
    94→  for (const s of sales) {
    95→    const t = new Date(s.createdAt);
    96→    t.setHours(0, 0, 0, 0);
    97→    const idx = days.findIndex((d) => d.date.getTime() === t.getTime());
    98→    if (idx >= 0) days[idx].total += s.total;
    99→  }
   100→  return days;
   101→}
