   123→function CameraScannerBody({
   124→  onScan,
   125→  onOpenChange,
   126→}: {
   127→  onScan: (code: string) => void;
   128→  onOpenChange: (o: boolean) => void;
   129→}) {
   130→  const videoRef = useRef<HTMLVideoElement>(null);
   131→  const streamRef = useRef<MediaStream | null>(null);
   132→  const [cameraReady, setCameraReady] = useState(false);
   133→  const [cameraError, setCameraError] = useState<string | null>(null);
   134→  const [manualCode, setManualCode] = useState("");
   135→
   136→  useEffect(() => {
   137→    let cancelled = false;
   138→
   139→    async function startCamera() {
   140→      try {
   141→        if (!navigator.mediaDevices?.getUserMedia) {
   142→          setCameraError("Camera API not supported in this browser.");
   143→          return;
   144→        }
   145→        const stream = await navigator.mediaDevices.getUserMedia({
   146→          video: { facingMode: "environment" },
   147→          audio: false,
   148→        });
   149→        if (cancelled) {
   150→          stream.getTracks().forEach((t) => t.stop());
   151→          return;
   152→        }
   153→        streamRef.current = stream;
   154→        if (videoRef.current) {
   155→          videoRef.current.srcObject = stream;
   156→          await videoRef.current.play().catch(() => {
   157→            /* autoplay may be blocked — ignore */
   158→          });
   159→        }
   160→        if (!cancelled) setCameraReady(true);
   161→      } catch (err) {
   162→        if (cancelled) return;
   163→        const message =
   164→          err instanceof Error ? err.message : "Unable to access camera.";
   165→        setCameraError(message);
   166→      }
   167→    }
   168→
   169→    startCamera();
   170→
   171→    return () => {
   172→      cancelled = true;
   173→      if (streamRef.current) {
   174→        streamRef.current.getTracks().forEach((t) => t.stop());
   175→        streamRef.current = null;
   176→      }
   177→    };
   178→  }, []);
   179→
   180→  const submitManual = () => {
   181→    const code = manualCode.trim();
   182→    if (!code) return;
   183→    onScan(code);
   184→    setManualCode("");
   185→    onOpenChange(false);
   186→  };
   187→
   188→  return (
   189→    <>
   190→      <DialogHeader>
   191→        <DialogTitle className="flex items-center gap-2">
   192→          <ScanLine className="h-5 w-5 text-primary" />
   193→          Scan Barcode
   194→        </DialogTitle>
   195→        <DialogDescription>
   196→          Point your camera at a product barcode, or enter the code manually
   197→          below.
   198→        </DialogDescription>
   199→      </DialogHeader>
   200→
   201→      {/* Camera preview */}
   202→      <div className="relative aspect-video w-full overflow-hidden rounded-lg border bg-black">
   203→        {cameraError ? (
   204→          <div className="absolute inset-0 flex flex-col items-center justify-center gap-2 p-4 text-center">
   205→            <CameraOff className="h-8 w-8 text-muted-foreground" />
   206→            <p className="text-sm font-medium text-foreground">
   207→              Camera not available
   208→            </p>
   209→            <p className="text-xs text-muted-foreground">
   210→              Enter the barcode manually below to continue.
   211→            </p>
   212→          </div>
   213→        ) : (
   214→          <>
   215→            <video
   216→              ref={videoRef}
   217→              autoPlay
   218→              playsInline
   219→              muted
   220→              className="h-full w-full object-cover"
   221→            />
   222→            {/* Scan frame overlay */}
   223→            <div className="pointer-events-none absolute inset-0 flex items-center justify-center">
   224→              <div className="relative h-24 w-3/4 max-w-[260px] rounded-md border-2 border-primary/80 shadow-[0_0_0_9999px_rgba(0,0,0,0.35)]">
   225→                <span className="absolute -left-1 -top-1 h-4 w-4 border-l-4 border-t-4 border-primary" />
   226→                <span className="absolute -right-1 -top-1 h-4 w-4 border-r-4 border-t-4 border-primary" />
   227→                <span className="absolute -bottom-1 -left-1 h-4 w-4 border-b-4 border-l-4 border-primary" />
   228→                <span className="absolute -bottom-1 -right-1 h-4 w-4 border-b-4 border-r-4 border-primary" />
   229→                <motion.span
   230→                  className="absolute inset-x-1 top-1/2 h-0.5 bg-primary"
   231→                  initial={{ y: -40 }}
   232→                  animate={{ y: 40 }}
   233→                  transition={{
   234→                    duration: 1.6,
   235→                    repeat: Infinity,
   236→                    repeatType: "reverse",
   237→                    ease: "easeInOut",
   238→                  }}
   239→                />
   240→              </div>
   241→            </div>
   242→            {!cameraReady && !cameraError && (
   243→              <div className="absolute inset-0 flex items-center justify-center text-xs text-white/70">
   244→                Starting camera…
   245→              </div>
   246→            )}
   247→          </>
   248→        )}
   249→      </div>
   250→
   251→      {/* Manual fallback — primary path */}
   252→      <div className="space-y-2">
   253→        <label className="text-xs font-medium text-muted-foreground">
   254→          Enter barcode manually
   255→        </label>
   256→        <div className="flex gap-2">
   257→          <Input
   258→            autoFocus
   259→            value={manualCode}
   260→            placeholder="e.g. 8901030867531"
   261→            onChange={(e) => setManualCode(e.target.value)}
   262→            onKeyDown={(e) => {
   263→              if (e.key === "Enter") submitManual();
   264→            }}
   265→          />
   266→          <Button type="button" onClick={submitManual} disabled={!manualCode.trim()}>
   267→            <Plus className="h-4 w-4" />
   268→            Add
   269→          </Button>
   270→        </div>
   271→      </div>
   272→
   273→      <DialogFooter>
   274→        <Button variant="outline" onClick={() => onOpenChange(false)}>
   275→          Close
   276→        </Button>
   277→      </DialogFooter>
   278→    </>
   279→  );
   280→}

   286→function ProductTile({
   287→  product,
   288→  currency,
   289→  onAdd,
   290→}: {
   291→  product: Product;
   292→  currency: string;
   293→  onAdd: (p: Product) => void;
   294→}) {
   295→  const outOfStock = product.stock <= 0;
   296→  return (
   297→    <motion.button
   298→      type="button"
   299→      whileHover={outOfStock ? undefined : { y: -4, scale: 1.02 }}
   300→      whileTap={outOfStock ? undefined : { scale: 0.96 }}
   301→      transition={{ type: "spring", stiffness: 400, damping: 25 }}
   302→      disabled={outOfStock}
   303→      onClick={() => {
   304→        if (outOfStock) {
   305→          toast.error("Out of stock", { description: product.name });
   306→          return;
   307→        }
   308→        onAdd(product);
   309→      }}
   310→      className={cn(
   311→        "group relative text-left",
   312→        outOfStock && "cursor-not-allowed opacity-50 grayscale"
   313→      )}
   314→    >
   315→      <Card
   316→        className={cn(
   317→          "h-full gap-0 overflow-hidden p-0 transition-all duration-200",
   318→          outOfStock
   319→            ? "bg-muted/40"
   320→            : "group-hover:border-primary/50 group-hover:bg-accent/60 group-hover:shadow-lg group-hover:shadow-primary/10"
   321→        )}
   322→      >
   323→        <div
   324→          className="relative flex h-20 items-center justify-center text-xl font-bold transition-transform duration-200 group-hover:scale-105"
   325→          style={{ backgroundColor: colorFromString(product.name) }}
   326→        >
   327→          <span className="grid h-12 w-12 place-items-center rounded-full bg-white/70 text-sm font-bold text-foreground/80 backdrop-blur transition-transform duration-200 group-hover:scale-110">
   328→            {initialsOf(product.name)}
   329→          </span>
   330→          {/* Shine effect on hover */}
   331→          {!outOfStock && (
   332→            <span className="pointer-events-none absolute inset-0 -translate-x-full bg-gradient-to-r from-transparent via-white/20 to-transparent transition-transform duration-500 group-hover:translate-x-full" />
   333→          )}
   334→        </div>
   335→        <div className="space-y-1.5 p-2.5">
   336→          <p className="line-clamp-2 text-xs font-medium leading-snug text-foreground min-h-[2rem]">
   337→            {product.name}
   338→          </p>
   339→          <div className="flex items-center justify-between gap-1">
   340→            <span className="text-sm font-bold text-primary transition-colors group-hover:text-primary/80">
   341→              {formatMoney(product.price, currency)}
   342→            </span>
   343→            {stockBadge(product.stock)}
   344→          </div>
   345→        </div>
   346→      </Card>
   347→    </motion.button>
   348→  );
   349→}

   442→export function PosTerminal() {
   443→  const { data: products = [], isLoading: productsLoading } = useQuery({
   444→    queryKey: ["products"],
   445→    queryFn: productsApi.list,
   446→  });
   447→
   448→  const { data: shop } = useQuery({
   449→    queryKey: ["shop"],
   450→    queryFn: shopApi.get,
   451→  });
   452→
   453→  const currency = shop?.currency || "₹";
   454→  const taxRate = shop?.taxRate ?? 0;
   455→
   456→  // Cart store — select primitive slices to avoid re-render storms
   457→  const items = useCartStore((s) => s.items);
   458→  const discount = useCartStore((s) => s.discount);
   459→  const paymentMethod = useCartStore((s) => s.paymentMethod);
   460→  const addProduct = useCartStore((s) => s.addProduct);
   461→  const addByBarcode = useCartStore((s) => s.addByBarcode);
   462→  const removeItem = useCartStore((s) => s.removeItem);
   463→  const updateQuantity = useCartStore((s) => s.updateQuantity);
   464→  const clearCart = useCartStore((s) => s.clearCart);
   465→  const setDiscount = useCartStore((s) => s.setDiscount);
   466→  const setPaymentMethod = useCartStore((s) => s.setPaymentMethod);
   467→
   468→  // Held orders (parked sales) — persisted to localStorage
   469→  const heldOrders = useHeldOrdersStore((s) => s.orders);
   470→  const holdOrder = useHeldOrdersStore((s) => s.holdOrder);
   471→  const resumeOrder = useHeldOrdersStore((s) => s.resumeOrder);
   472→  const deleteHeldOrder = useHeldOrdersStore((s) => s.deleteOrder);
   473→
   474→  const [scanInput, setScanInput] = useState("");
   475→  const [activeCategory, setActiveCategory] = useState<string>("All");
   476→  const [searchTerm, setSearchTerm] = useState("");
   477→  const [scannerOpen, setScannerOpen] = useState(false);
   478→  const [checkoutOpen, setCheckoutOpen] = useState(false);
   479→  const [clearConfirmOpen, setClearConfirmOpen] = useState(false);
   480→  const [shortcutsHintOpen, setShortcutsHintOpen] = useState(false);
   481→  const [heldOrdersOpen, setHeldOrdersOpen] = useState(false);
   482→  const [customItemOpen, setCustomItemOpen] = useState(false);
   483→  const [customName, setCustomName] = useState("");
   484→  const [customPrice, setCustomPrice] = useState("");
   485→
   486→  const scanInputRef = useRef<HTMLInputElement>(null);
   487→
   488→  const handleAddCustomItem = () => {
   489→    const price = Number(customPrice);
   490→    if (!customPrice || Number.isNaN(price) || price <= 0) {
   491→      toast.error("Enter a valid price");
   492→      return;
   493→    }
   494→    const name = customName.trim() || "Custom Item";
   495→    const customProduct: Product = {
   496→      id: `custom-${Date.now()}`,
   497→      name,
   498→      barcode: `CUSTOM-${Date.now().toString().slice(-6)}`,
   499→      price,
   500→      costPrice: 0,
   501→      stock: 999999,
   502→      category: "Custom",
   503→      createdAt: new Date().toISOString(),
   504→      updatedAt: new Date().toISOString(),
   505→    };
   506→    addProduct(customProduct);
   507→    toast.success(`${name} added`, { description: formatMoney(price, currency) });
   508→    setCustomName("");
   509→    setCustomPrice("");
   510→    setCustomItemOpen(false);
   511→  };
   512→
   513→  const handleHoldOrder = () => {
   514→    if (items.length === 0) return;
   515→    holdOrder({ items, discount, paymentMethod });
   516→    const count = items.reduce((s, i) => s + i.quantity, 0);
   517→    toast.success(`Order held (${count} items)`, {
   518→      description: "Press F7 to resume held orders",
   519→      duration: 2500,
   520→    });
   521→    clearCart();
   522→  };
   523→
   524→  const handleResumeOrder = (id: string) => {
   525→    const order = resumeOrder(id);
   526→    if (!order) {
   527→      toast.error("Order not found");
   528→      return;
   529→    }
   530→    // If current cart is non-empty, hold it first
   531→    if (items.length > 0) {
   532→      holdOrder({ items, discount, paymentMethod });
   533→      toast.info("Current cart held", { duration: 1500 });
   534→    }
   535→    // Restore the held order into the cart
   536→    order.items.forEach((ci) => {
   537→      for (let i = 0; i < ci.quantity; i++) {
   538→        addProduct(ci.product);
   539→      }
   540→    });
   541→    setDiscount(order.discount);
   542→    setPaymentMethod(order.paymentMethod);
   543→    setHeldOrdersOpen(false);
   544→    toast.success("Order resumed", { duration: 1500 });
   545→  };
   546→
   547→  const handleDeleteHeldOrder = (id: string) => {
   548→    deleteHeldOrder(id);
   549→    toast.success("Held order discarded");
   550→  };
   551→
   552→  // Keyboard shortcuts: F1 focus scan, F2/F3/F4 payment methods, F6 hold, F7 held, F9 checkout
   553→  useEffect(() => {
   554→    const handler = (e: KeyboardEvent) => {
   555→      // Skip if a dialog is open or user is typing in a non-scan input
   556→      const target = e.target as HTMLElement | null;
   557→      const isTyping =
   558→        target?.tagName === "INPUT" ||
   559→        target?.tagName === "TEXTAREA" ||
   560→        target?.isContentEditable;
   561→
   562→      // F1 — focus scan input (always works, even if typing)
   563→      if (e.key === "F1") {
   564→        e.preventDefault();
   565→        scanInputRef.current?.focus();
   566→        scanInputRef.current?.select();
   567→        return;
   568→      }
   569→
   570→      // F2/F3/F4 — payment method selection (only when cart non-empty)
   571→      if (e.key === "F2" && items.length > 0) {
   572→        e.preventDefault();
   573→        setPaymentMethod("Cash");
   574→        toast.success("Payment: Cash", { duration: 1000 });
   575→        return;
   576→      }
   577→      if (e.key === "F3" && items.length > 0) {
   578→        e.preventDefault();
   579→        setPaymentMethod("UPI");
   580→        toast.success("Payment: UPI", { duration: 1000 });
   581→        return;
   582→      }
   583→      if (e.key === "F4" && items.length > 0) {
   584→        e.preventDefault();
   585→        setPaymentMethod("Card");
   586→        toast.success("Payment: Card", { duration: 1000 });
   587→        return;
   588→      }
   589→
   590→      // F9 — checkout (only when cart non-empty and not typing in scan input)
   591→      if (e.key === "F9" && items.length > 0 && !isTyping) {
   592→        e.preventDefault();
   593→        setCheckoutOpen(true);
   594→        return;
   595→      }
   596→
   597→      // F6 — hold current order (only when cart non-empty and not typing)
   598→      if (e.key === "F6" && items.length > 0 && !isTyping) {
   599→        e.preventDefault();
   600→        handleHoldOrder();
