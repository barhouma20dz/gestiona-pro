    75→function stockBadge(stock: number, t: (k: "outOfStock" | "left" | "inStock" | "lowWithLeft") => string) {
    76→  if (stock === 0) {
    77→    return (
    78→      <Badge className="bg-red-500/15 text-red-600 border-red-500/30 hover:bg-red-500/15">
    79→        {t("outOfStock")}
    80→      </Badge>
    81→    );
    82→  }
    83→  if (stock <= 5) {
    84→    return (
    85→      <Badge className="bg-amber-500/15 text-amber-600 border-amber-500/30 hover:bg-amber-500/15">
    86→        {t("lowWithLeft")} {stock} {t("left")}
    87→      </Badge>
    88→    );
    89→  }
    90→  return (
    91→    <Badge className="bg-emerald-500/15 text-emerald-600 border-emerald-500/30 hover:bg-emerald-500/15">
    92→      {stock} {t("inStock")}
    93→    </Badge>
    94→  );
    95→}
    96→
    97→function categoryBadge(category: string) {
    98→  return (
    99→    <Badge variant="outline" className="font-medium">
   100→      {category}
   101→    </Badge>
   102→  );
   103→}

   131→function toForm(p: Product): FormState {
   132→  return {
   133→    name: p.name,
   134→    barcode: p.barcode,
   135→    price: String(p.price),
   136→    costPrice: String(p.costPrice ?? 0),
   137→    stock: String(p.stock),
   138→    category: p.category,
   139→    expiryDate: p.expiryDate ? new Date(p.expiryDate).toISOString().split("T")[0] : "",
   140→    expiryDate2: p.expiryDate2 ? new Date(p.expiryDate2).toISOString().split("T")[0] : "",

   147→export function ProductManager() {

   156→    queryFn: shopApi.get,
   157→  });

   159→
   160→  // Filter state
   161→  const [search, setSearch] = useState("");
   162→  const [category, setCategory] = useState<string>("all");
   163→
   164→  // Dialog state
   165→  const [formOpen, setFormOpen] = useState(false);
   166→  const [editing, setEditing] = useState<Product | null>(null);
   167→  const [scannerOpen, setScannerOpen] = useState(false);
   168→  const fileInputRef = useRef<HTMLInputElement>(null);
   169→  const cameraInputRef = useRef<HTMLInputElement>(null);
   170→
   171→  // Convert file to base64 data URL for storage
   172→  const handleFileUpload = (file: File) => {
   173→    if (!file) return;
   174→    if (file.size > 2 * 1024 * 1024) {
   175→      toast.error("حجم الصورة كبير جداً (الحد 2MB)");
   176→      return;
   177→    }
   178→    const reader = new FileReader();
   179→    reader.onload = () => {
   180→      const result = reader.result as string;
   181→      setForm((f) => ({ ...f, imageUrl: result }));
   182→    };
   183→    reader.readAsDataURL(file);
   184→  };
   185→  const [form, setForm] = useState<FormState>(emptyForm);
   186→  const [formErrors, setFormErrors] = useState<Partial<Record<keyof FormState, string>>>({});
   187→
   188→  // Delete state
   189→  const [deleteTarget, setDeleteTarget] = useState<Product | null>(null);
   190→
   191→  /* Derived stats */
   192→  const stats = useMemo(() => {
   193→    const list = products ?? [];
   194→    const totalValue = list.reduce((sum, p) => sum + p.price * p.stock, 0);
   195→    const totalCost = list.reduce(
   196→      (sum, p) => sum + (p.costPrice ?? 0) * p.stock,
   197→      0
   198→    );
   199→    const potentialProfit = totalValue - totalCost;
   200→    const avgMargin =
   201→      totalValue > 0 ? (potentialProfit / totalValue) * 100 : 0;
   202→    const lowStock = list.filter((p) => p.stock >= 1 && p.stock <= 5).length;
   203→    const outStock = list.filter((p) => p.stock === 0).length;
   204→    return {
   205→      total: list.length,
   206→      totalValue,
   207→      totalCost,
   208→      potentialProfit,
   209→      avgMargin,
   210→      lowStock,
   211→      outStock,
   212→    };
   213→  }, [products]);
   214→
   215→  /* Unique categories for filter + datalist */
   216→  const categories = useMemo(() => {
   217→    const set = new Set<string>();
   218→    (products ?? []).forEach((p) => set.add(p.category));
   219→    return Array.from(set).sort((a, b) => a.localeCompare(b));
   220→  }, [products]);
   221→
   222→  /* Filtered list */
   223→  const filtered = useMemo(() => {
   224→    const list = products ?? [];
   225→    const q = search.trim().toLowerCase();
   226→    return list
   227→      .filter((p) => (category === "all" ? true : p.category === category))
   228→      .filter((p) =>
   229→        q === ""
   230→          ? true
   231→          : p.name.toLowerCase().includes(q) ||
   232→            p.barcode.toLowerCase().includes(q)
   233→      )
   234→      .sort(
   235→        (a, b) =>
   236→          new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   237→      );
   238→  }, [products, search, category]);
   239→
   240→  /* Mutations */
   241→  const createMutation = useMutation({
   242→    mutationFn: (data: NewProduct) => productsApi.create(data),
   243→    onSuccess: () => {
   244→      toast.success(t("productCreated"));
   245→      qc.invalidateQueries({ queryKey: ["products"] });
   246→      setFormOpen(false);
   247→    },
   248→    onError: (e: Error) => toast.error(e.message || t("failedCreateProduct")),
   249→  });
   250→
   251→  const updateMutation = useMutation({
   252→    mutationFn: ({ id, data }: { id: string; data: Partial<NewProduct> }) =>
   253→      productsApi.update(id, data),
   254→    onSuccess: () => {
   255→      toast.success(t("productUpdated"));

   257→      setFormOpen(false);
   258→    },
   259→    onError: (e: Error) => toast.error(e.message || t("failedUpdateProduct")),
   260→  });
   261→

   263→    mutationFn: (id: string) => productsApi.delete(id),
   264→    onSuccess: () => {
   265→      toast.success(t("productDeleted"));
   266→      qc.invalidateQueries({ queryKey: ["products"] });
   267→      setDeleteTarget(null);
   268→    },
   269→    onError: (e: Error) => toast.error(e.message || t("failedDeleteProduct")),
   270→  });
   271→
   272→  /* Form handlers */
   273→  const openAdd = () => {
   274→    setEditing(null);
   275→    setForm(emptyForm);
   276→    setFormErrors({});
   277→    setFormOpen(true);
   278→  };
   279→

   284→    setFormOpen(true);
   285→  };
   286→

   289→    setEditing(null);
   290→    setForm({
   291→      name: `${p.name} (Copy)`,
   292→      barcode: `${p.barcode}-COPY`,
   293→      price: String(p.price),
   294→      stock: String(p.stock),
   295→      category: p.category,
   296→    });
   297→    setFormErrors({});
   298→    setFormOpen(true);
   299→    toast.info(t("productDuplicated"), { duration: 2000 });
   300→  };
   301→
   302→  const validate = (): boolean => {
   303→    const errs: Partial<Record<keyof FormState, string>> = {};
   304→    if (!form.name.trim()) errs.name = t("nameRequired");
   305→    if (!form.barcode.trim()) errs.barcode = t("barcodeRequired");
   306→    const price = Number(form.price);
   307→    if (form.price === "" || Number.isNaN(price) || price < 0)
   308→      errs.price = t("priceMin");
   309→    const costPrice = Number(form.costPrice);
   310→    if (form.costPrice === "" || Number.isNaN(costPrice) || costPrice < 0)
   311→      errs.costPrice = t("costMin");
   312→    const stock = Number(form.stock);
   313→    if (form.stock === "" || Number.isNaN(stock) || stock < 0)
   314→      errs.stock = t("stockMin");
   315→    setFormErrors(errs);
   316→    return Object.keys(errs).length === 0;
   317→  };
   318→
   319→  const handleSubmit = (e: React.FormEvent) => {
   320→    e.preventDefault();
   321→    if (!validate()) return;
   322→    const payload: NewProduct = {
   323→      name: form.name.trim(),
   324→      barcode: form.barcode.trim(),
   325→      price: Number(form.price),
   326→      costPrice: Number(form.costPrice),
   327→      stock: Number(form.stock),
   328→      category: form.category.trim() || "General",
   329→      expiryDate: form.expiryDate || null,
   330→      expiryDate2: form.expiryDate2 || null,
   331→      imageUrl: form.imageUrl || null,
   332→    };
   333→    if (editing) {
   334→      updateMutation.mutate({ id: editing.id, data: payload });
   335→    } else {
   336→      createMutation.mutate(payload);
   337→    }
   338→  };
   339→
   340→  const saving = createMutation.isPending || updateMutation.isPending;
   341→
   342→  const statCards = [
   343→    {
   344→      label: t("totalProducts"),
   345→      value: String(stats.total),
   346→      icon: Package,
   347→      tint: "bg-emerald-500/15 text-emerald-600",
   348→    },
   349→    {
   350→      label: t("inventoryValue"),
   351→      value: formatMoney(stats.totalValue, currency),
   352→      icon: Wallet,
   353→      tint: "bg-emerald-500/15 text-emerald-600",
   354→    },
   355→    {
   356→      label: t("potentialProfit"),
   357→      value: formatMoney(stats.potentialProfit, currency),
   358→      sub: `${stats.avgMargin.toFixed(1)}% ${t("margin")}`,
   359→      icon: TrendingUp,
   360→      tint: "bg-emerald-500/15 text-emerald-600",
   361→    },
   362→    {
   363→      label: t("lowOutStock"),
   364→      value: `${stats.lowStock} / ${stats.outStock}`,
   365→      icon: AlertTriangle,
   366→      tint: "bg-amber-500/15 text-amber-600",
   367→    },
   368→  ];
   369→
   370→  return (
   371→    <div className="space-y-6">
   372→      {/* Gradient page header */}
   373→      <div className="relative overflow-hidden rounded-2xl border bg-gradient-to-br from-emerald-600 to-teal-700 p-5 text-white shadow-lg sm:p-6">
   374→        <div className="pointer-events-none absolute -right-12 -top-12 h-40 w-40 rounded-full bg-white/10 blur-2xl" />
   375→        <div className="relative flex flex-wrap items-center justify-between gap-3">
   376→          <div>
   377→            <h2 className="text-lg font-bold tracking-tight drop-shadow-sm sm:text-xl">
   378→              {t("inventoryManagement")}
   379→            </h2>
   380→            <p className="mt-0.5 text-sm text-emerald-50/80">
   381→              {stats.total} {t("productsLabel")} · {formatMoney(stats.totalValue, currency)} {t("inventoryValue").toLowerCase()}
   382→            </p>
   383→          </div>
   384→          <Button
   385→            onClick={openAdd}
   386→            className="bg-white text-emerald-700 shadow-md hover:bg-emerald-50"
   387→          >
   388→            <Plus className="h-4 w-4" />
   389→            {t("addProduct")}
   390→          </Button>
   391→        </div>
   392→      </div>
   393→
   394→      {/* Stats row */}
   395→      <div className="grid grid-cols-2 gap-3 sm:gap-4 lg:grid-cols-4">
   396→        {statCards.map((s, i) => {
   397→          const Icon = s.icon;
   398→          return (
   399→            <motion.div
   400→              key={s.label}
   401→              initial={{ opacity: 0, y: 8 }}
   402→              animate={{ opacity: 1, y: 0 }}
   403→              transition={{ delay: i * 0.05, duration: 0.25 }}
   404→            >
   405→              <Card className="gap-0 p-4 sm:p-5">
   406→                <div className="flex items-start justify-between gap-3">
   407→                  <div className="min-w-0">
   408→                    <p className="text-xs font-medium text-muted-foreground">
   409→                      {s.label}
   410→                    </p>
   411→                    <p className="mt-1 text-xl font-bold tracking-tight sm:text-2xl">
   412→                      {s.value}
   413→                    </p>
   414→                    {s.sub && (
   415→                      <p className="mt-0.5 text-[11px] font-medium text-emerald-600">
   416→                        {s.sub}
   417→                      </p>
   418→                    )}
   419→                  </div>
   420→                  <div
   421→                    className={cn(
   422→                      "grid size-10 shrink-0 place-items-center rounded-xl",
   423→                      s.tint
   424→                    )}
   425→                  >
   426→                    <Icon className="h-5 w-5" />
   427→                  </div>
   428→                </div>
   429→              </Card>
   430→            </motion.div>
   431→          );
   432→        })}
   433→      </div>
   434→
   435→      {/* Toolbar */}
   436→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   437→        <div className="relative flex-1">
   438→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   439→          <Input
   440→            placeholder={t("searchByNameOrBarcode")}
   441→            value={search}
   442→            onChange={(e) => setSearch(e.target.value)}
   443→            className="pl-9"
   444→          />
   445→        </div>
   446→        <Select value={category} onValueChange={setCategory}>
   447→          <SelectTrigger className="w-full sm:w-[180px]">
   448→            <SelectValue placeholder={t("category")} />
   449→          </SelectTrigger>
   450→          <SelectContent>
   451→            <SelectItem value="all">{t("allCategories")}</SelectItem>
   452→            {categories.map((c) => (
   453→              <SelectItem key={c} value={c}>
   454→                {c}
   455→              </SelectItem>
   456→            ))}
   457→          </SelectContent>
   458→        </Select>
   459→        <Button onClick={openAdd} className="w-full sm:w-auto">
   460→          <Plus className="h-4 w-4" />
   461→          {t("addProduct")}
   462→        </Button>
   463→      </div>
   464→
   465→      {/* Products card */}
   466→      <Card className="gap-0 p-0">
   467→        <div className="flex items-center justify-between border-b px-4 py-3 sm:px-6">
   468→          <h2 className="text-sm font-semibold">
   469→            {t("productsLabel")}{" "}
   470→            <span className="text-muted-foreground">({filtered.length})</span>
   471→          </h2>
   472→          {filtered.length > 0 && (
   473→            <span className="hidden text-xs text-muted-foreground sm:inline">
   474→              {t("sortedByNewest")}
   475→            </span>
   476→          )}
   477→        </div>
   478→
   479→        {/* Desktop / tablet table */}
   480→        <div className="hidden md:block">
   481→          <Table>
   482→            <TableHeader>
   483→              <TableRow className="bg-muted/40 hover:bg-muted/40">
   484→                <TableHead className="pl-6">{t("product")}</TableHead>
   485→                <TableHead>{t("category")}</TableHead>
   486→                <TableHead className="text-right">{t("priceLabel")}</TableHead>
   487→                <TableHead>{t("stock")}</TableHead>
   488→                <TableHead className="pr-6 text-right">{t("actions")}</TableHead>

   493→                <TableRow>

   495→                    <div className="flex items-center justify-center gap-2 text-muted-foreground">
   496→                      <Loader2 className="h-4 w-4 animate-spin" />
   497→                      {t("loadingProducts")}
   498→                    </div>
   499→                  </TableCell>
   500→                </TableRow>
   501→              ) : filtered.length === 0 ? (
   502→                <TableRow>
   503→                  <TableCell colSpan={5} className="py-12 text-center">
   504→                    <EmptyState onAdd={openAdd} hasProducts={(products?.length ?? 0) > 0} t={t} />
   505→                  </TableCell>
   506→                </TableRow>
   507→              ) : (
   508→                filtered.map((p) => (
   509→                  <TableRow
   510→                    key={p.id}
   511→                    className="group transition-colors hover:bg-emerald-500/5"
   512→                  >
   513→                    <TableCell className="pl-6 py-3.5">
   514→                      <div className="flex flex-col">
   515→                        <span className="font-medium transition-colors group-hover:text-primary">
   516→                          {p.name}
   517→                        </span>
   518→                        <span className="font-mono text-[11px] text-muted-foreground">
   519→                          {p.barcode}
   520→                        </span>
   521→                      </div>
   522→                    </TableCell>
   523→                    <TableCell>{categoryBadge(p.category)}</TableCell>
   524→                    <TableCell className="text-right font-medium tabular-nums">
   525→                      {formatMoney(p.price, currency)}
   526→                    </TableCell>
   527→                    <TableCell>{stockBadge(p.stock, t)}</TableCell>
   528→                    <TableCell className="pr-6 text-right">
   529→                      <div className="flex items-center justify-end gap-1">
   530→                        <Button
   531→                          variant="ghost"
   532→                          size="icon"
   533→                          className="h-8 w-8"
   534→                          onClick={() => openEdit(p)}
   535→                          aria-label={`${t("edit")} ${p.name}`}
   536→                        >
   537→                          <Pencil className="h-4 w-4" />
   538→                        </Button>
   539→                        <Button
   540→                          variant="ghost"
   541→                          size="icon"
   542→                          className="h-8 w-8 text-muted-foreground hover:text-primary"
   543→                          onClick={() => duplicateProduct(p)}
   544→                          aria-label={`${t("duplicate")} ${p.name}`}
   545→                          title={t("duplicateProduct")}
   546→                        >
   547→                          <Copy className="h-4 w-4" />
   548→                        </Button>
   549→                        <Button
   550→                          variant="ghost"
   551→                          size="icon"
   552→                          className="h-8 w-8 text-red-600 hover:bg-red-500/10 hover:text-red-700"
   553→                          onClick={() => setDeleteTarget(p)}
   554→                          aria-label={`${t("delete")} ${p.name}`}
   555→                        >
   556→                          <Trash2 className="h-4 w-4" />
   557→                        </Button>
   558→                      </div>
   559→                    </TableCell>
   560→                  </TableRow>
   561→                ))
   562→              )}
   563→            </TableBody>
   564→          </Table>
   565→        </div>
   566→
   567→        {/* Mobile card list */}
   568→        <div className="block md:hidden">
   569→          {isLoading ? (
   570→            <div className="flex items-center justify-center gap-2 py-12 text-muted-foreground">
   571→              <Loader2 className="h-4 w-4 animate-spin" />
   572→              {t("loadingProducts")}
   573→            </div>
   574→          ) : filtered.length === 0 ? (
   575→            <div className="px-4 py-12">
   576→              <EmptyState onAdd={openAdd} hasProducts={(products?.length ?? 0) > 0} t={t} />
   577→            </div>
   578→          ) : (
   579→            <ul className="divide-y">
   580→              {filtered.map((p) => (
   581→                <li key={p.id} className="p-4">
   582→                  <div className="flex items-start justify-between gap-3">
