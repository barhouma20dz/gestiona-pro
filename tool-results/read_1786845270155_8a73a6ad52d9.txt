   125→function CameraScannerBody({
   126→  onScan,
   127→  onOpenChange,
   128→}: {
   129→  onScan: (code: string) => void;
   130→  onOpenChange: (o: boolean) => void;
   131→}) {
   132→  const { t } = useLanguage();
   133→  const videoRef = useRef<HTMLVideoElement>(null);
   134→  const streamRef = useRef<MediaStream | null>(null);
   135→  const [cameraReady, setCameraReady] = useState(false);
   136→  const [cameraError, setCameraError] = useState<string | null>(null);
   137→  const [manualCode, setManualCode] = useState("");
   138→
   139→  useEffect(() => {
   140→    let cancelled = false;
   141→
   142→    async function startCamera() {
   143→      try {
   144→        if (!navigator.mediaDevices?.getUserMedia) {
   145→          setCameraError("Camera API not supported in this browser.");
   146→          return;
   147→        }
   148→        const stream = await navigator.mediaDevices.getUserMedia({
   149→          video: { facingMode: "environment" },
   150→          audio: false,
   151→        });
   152→        if (cancelled) {
   153→          stream.getTracks().forEach((t) => t.stop());
   154→          return;
   155→        }
   156→        streamRef.current = stream;
   157→        if (videoRef.current) {
   158→          videoRef.current.srcObject = stream;
   159→          await videoRef.current.play().catch(() => {
   160→            /* autoplay may be blocked — ignore */
   161→          });
   162→        }
   163→        if (!cancelled) setCameraReady(true);
   164→      } catch (err) {
   165→        if (cancelled) return;
   166→        const message =
   167→          err instanceof Error ? err.message : "Unable to access camera.";
   168→        setCameraError(message);
   169→      }
   170→    }
   171→
   172→    startCamera();
   173→
   174→    return () => {
   175→      cancelled = true;
   176→      if (streamRef.current) {
   177→        streamRef.current.getTracks().forEach((t) => t.stop());
   178→        streamRef.current = null;
   179→      }
   180→    };
   181→  }, []);
   182→
   183→  const submitManual = () => {
   184→    const code = manualCode.trim();
   185→    if (!code) return;
   186→    onScan(code);
   187→    setManualCode("");
   188→    onOpenChange(false);
   189→  };
   190→
   191→  return (
   192→    <>
   193→      <DialogHeader>
   194→        <DialogTitle className="flex items-center gap-2">
   195→          <ScanLine className="h-5 w-5 text-primary" />
   196→          {t("scanBarcode")}
   197→        </DialogTitle>
   198→        <DialogDescription>
   199→          {t("cameraHint")}
   200→        </DialogDescription>
   201→      </DialogHeader>
   202→
   203→      {/* Camera preview */}
   204→      <div className="relative aspect-video w-full overflow-hidden rounded-lg border bg-black">
   205→        {cameraError ? (
   206→          <div className="absolute inset-0 flex flex-col items-center justify-center gap-2 p-4 text-center">
   207→            <CameraOff className="h-8 w-8 text-muted-foreground" />
   208→            <p className="text-sm font-medium text-foreground">
   209→              {t("cameraNotAvailable")}
   210→            </p>
   211→            <p className="text-xs text-muted-foreground">
   212→              {t("manualBarcodeHint")}
   213→            </p>
   214→          </div>
   215→        ) : (
   216→          <>
   217→            <video
   218→              ref={videoRef}
   219→              autoPlay
   220→              playsInline
   221→              muted
   222→              className="h-full w-full object-cover"
   223→            />
   224→            {/* Scan frame overlay */}
   225→            <div className="pointer-events-none absolute inset-0 flex items-center justify-center">
   226→              <div className="relative h-24 w-3/4 max-w-[260px] rounded-md border-2 border-primary/80 shadow-[0_0_0_9999px_rgba(0,0,0,0.35)]">
   227→                <span className="absolute -left-1 -top-1 h-4 w-4 border-l-4 border-t-4 border-primary" />
   228→                <span className="absolute -right-1 -top-1 h-4 w-4 border-r-4 border-t-4 border-primary" />
   229→                <span className="absolute -bottom-1 -left-1 h-4 w-4 border-b-4 border-l-4 border-primary" />
   230→                <span className="absolute -bottom-1 -right-1 h-4 w-4 border-b-4 border-r-4 border-primary" />
   231→                <motion.span
   232→                  className="absolute inset-x-1 top-1/2 h-0.5 bg-primary"
   233→                  initial={{ y: -40 }}
   234→                  animate={{ y: 40 }}
   235→                  transition={{
   236→                    duration: 1.6,
   237→                    repeat: Infinity,
   238→                    repeatType: "reverse",
   239→                    ease: "easeInOut",
   240→                  }}
   241→                />
   242→              </div>
   243→            </div>
   244→            {!cameraReady && !cameraError && (
   245→              <div className="absolute inset-0 flex items-center justify-center text-xs text-white/70">
   246→                {t("startingCamera")}
   247→              </div>
   248→            )}
   249→          </>
   250→        )}
   251→      </div>
   252→
   253→      {/* Manual fallback — primary path */}
   254→      <div className="space-y-2">
   255→        <label className="text-xs font-medium text-muted-foreground">
   256→          {t("enterBarcodeManually")}
   257→        </label>
   258→        <div className="flex gap-2">
   259→          <Input
   260→            autoFocus
   261→            value={manualCode}
   262→            placeholder="e.g. 8901030867531"
   263→            onChange={(e) => setManualCode(e.target.value)}
   264→            onKeyDown={(e) => {
   265→              if (e.key === "Enter") submitManual();
   266→            }}
   267→          />
   268→          <Button type="button" onClick={submitManual} disabled={!manualCode.trim()}>
   269→            <Plus className="h-4 w-4" />
   270→            {t("add")}
   271→          </Button>
   272→        </div>
   273→      </div>
   274→
   275→      <DialogFooter>
   276→        <Button variant="outline" onClick={() => onOpenChange(false)}>
   277→          {t("close")}
   278→        </Button>
   279→      </DialogFooter>
   280→    </>
   281→  );
   282→}

   288→function ProductTile({
   289→  product,
   290→  currency,
   291→  onAdd,
   292→  t,
   293→}: {
   294→  product: Product;
   295→  currency: string;
   296→  onAdd: (p: Product) => void;
   297→  t: (k: "out" | "left" | "inStock" | "outOfStockMsg") => string;
   298→}) {
   299→  const outOfStock = product.stock <= 0;
   300→  return (
   301→    <motion.button
   302→      type="button"
   303→      whileHover={outOfStock ? undefined : { y: -4, scale: 1.02 }}
   304→      whileTap={outOfStock ? undefined : { scale: 0.96 }}
   305→      transition={{ type: "spring", stiffness: 400, damping: 25 }}
   306→      disabled={outOfStock}
   307→      onClick={() => {
   308→        if (outOfStock) {
   309→          toast.error(t("outOfStockMsg"), { description: product.name });
   310→          return;
   311→        }
   312→        onAdd(product);
   313→      }}
   314→      className={cn(
   315→        "group relative text-left",
   316→        outOfStock && "cursor-not-allowed opacity-50 grayscale"
   317→      )}
   318→    >
   319→      <Card
   320→        className={cn(
   321→          "h-full gap-0 overflow-hidden p-0 transition-all duration-200",
   322→          outOfStock
   323→            ? "bg-muted/40"
   324→            : "group-hover:border-primary/50 group-hover:bg-accent/60 group-hover:shadow-lg group-hover:shadow-primary/10"
   325→        )}
   326→      >
   327→        <div
   328→          className="relative flex h-20 items-center justify-center text-xl font-bold transition-transform duration-200 group-hover:scale-105"
   329→          style={{ backgroundColor: colorFromString(product.name) }}
   330→        >
   331→          <span className="grid h-12 w-12 place-items-center rounded-full bg-white/70 text-sm font-bold text-foreground/80 backdrop-blur transition-transform duration-200 group-hover:scale-110">
   332→            {initialsOf(product.name)}
   333→          </span>
   334→          {/* Shine effect on hover */}
   335→          {!outOfStock && (
   336→            <span className="pointer-events-none absolute inset-0 -translate-x-full bg-gradient-to-r from-transparent via-white/20 to-transparent transition-transform duration-500 group-hover:translate-x-full" />
   337→          )}
   338→        </div>
   339→        <div className="space-y-1.5 p-2.5">
   340→          <p className="line-clamp-2 text-xs font-medium leading-snug text-foreground min-h-[2rem]">
   341→            {product.name}
   342→          </p>
   343→          <div className="flex items-center justify-between gap-1">
   344→            <span className="text-sm font-bold text-primary transition-colors group-hover:text-primary/80">
   345→              {formatMoney(product.price, currency)}
   346→            </span>
   347→            {stockBadge(product.stock, t)}
   348→          </div>
   349→        </div>
   350→      </Card>
   351→    </motion.button>
   352→  );
   353→}

   448→export function PosTerminal() {
   449→  const { t } = useLanguage();
   450→  const { data: products = [], isLoading: productsLoading } = useQuery({
   451→    queryKey: ["products"],
   452→    queryFn: productsApi.list,
   453→  });
   454→
   455→  const { data: shop } = useQuery({
   456→    queryKey: ["shop"],
   457→    queryFn: shopApi.get,
   458→  });
   459→
   460→  const currency = shop?.currency || "₹";
   461→  const taxRate = shop?.taxRate ?? 0;
   462→
   463→  // Cart store — select primitive slices to avoid re-render storms
   464→  const items = useCartStore((s) => s.items);
   465→  const discount = useCartStore((s) => s.discount);
   466→  const paymentMethod = useCartStore((s) => s.paymentMethod);
   467→  const addProduct = useCartStore((s) => s.addProduct);
   468→  const addByBarcode = useCartStore((s) => s.addByBarcode);
   469→  const removeItem = useCartStore((s) => s.removeItem);
   470→  const updateQuantity = useCartStore((s) => s.updateQuantity);
   471→  const clearCart = useCartStore((s) => s.clearCart);
   472→  const setDiscount = useCartStore((s) => s.setDiscount);
   473→  const setPaymentMethod = useCartStore((s) => s.setPaymentMethod);
   474→
   475→  // Held orders (parked sales) — persisted to localStorage
   476→  const heldOrders = useHeldOrdersStore((s) => s.orders);
   477→  const holdOrder = useHeldOrdersStore((s) => s.holdOrder);
   478→  const resumeOrder = useHeldOrdersStore((s) => s.resumeOrder);
   479→  const deleteHeldOrder = useHeldOrdersStore((s) => s.deleteOrder);
   480→
   481→  const [scanInput, setScanInput] = useState("");
   482→  const [activeCategory, setActiveCategory] = useState<string>("All");
   483→  const [searchTerm, setSearchTerm] = useState("");
   484→  const [scannerOpen, setScannerOpen] = useState(false);
   485→  const [checkoutOpen, setCheckoutOpen] = useState(false);
   486→  const [clearConfirmOpen, setClearConfirmOpen] = useState(false);
   487→  const [shortcutsHintOpen, setShortcutsHintOpen] = useState(false);
   488→  const [heldOrdersOpen, setHeldOrdersOpen] = useState(false);
   489→  const [customItemOpen, setCustomItemOpen] = useState(false);
   490→  const [customName, setCustomName] = useState("");
   491→  const [customPrice, setCustomPrice] = useState("");
   492→
   493→  const scanInputRef = useRef<HTMLInputElement>(null);
   494→
   495→  const handleAddCustomItem = () => {
   496→    const price = Number(customPrice);
   497→    if (!customPrice || Number.isNaN(price) || price <= 0) {
   498→      toast.error(t("enterValidPrice"));
   499→      return;
   500→    }
   501→    const name = customName.trim() || t("customItemDefault");
   502→    const customProduct: Product = {
   503→      id: `custom-${Date.now()}`,
   504→      name,
   505→      barcode: `CUSTOM-${Date.now().toString().slice(-6)}`,
   506→      price,
   507→      costPrice: 0,
   508→      stock: 999999,
   509→      category: "Custom",
   510→      createdAt: new Date().toISOString(),
   511→      updatedAt: new Date().toISOString(),
   512→    };
   513→    addProduct(customProduct);
   514→    toast.success(`${name} ${t("productAdded")}`, { description: formatMoney(price, currency) });
   515→    setCustomName("");
   516→    setCustomPrice("");
   517→    setCustomItemOpen(false);
   518→  };
   519→
   520→  const handleHoldOrder = () => {
   521→    if (items.length === 0) return;
   522→    holdOrder({ items, discount, paymentMethod });
   523→    const count = items.reduce((s, i) => s + i.quantity, 0);
   524→    toast.success(`${t("orderHeld")} (${count} ${t("orderHeldItems")})`, {
   525→      description: t("pressF7Resume"),
   526→      duration: 2500,
   527→    });
   528→    clearCart();
   529→  };
   530→
   531→  const handleResumeOrder = (id: string) => {
   532→    const order = resumeOrder(id);
   533→    if (!order) {
   534→      toast.error(t("orderNotFound"));
   535→      return;
   536→    }
   537→    // If current cart is non-empty, hold it first
   538→    if (items.length > 0) {
   539→      holdOrder({ items, discount, paymentMethod });
   540→      toast.info(t("currentCartHeld"), { duration: 1500 });
   541→    }
   542→    // Restore the held order into the cart
   543→    order.items.forEach((ci) => {
   544→      for (let i = 0; i < ci.quantity; i++) {
   545→        addProduct(ci.product);
   546→      }
   547→    });
   548→    setDiscount(order.discount);
   549→    setPaymentMethod(order.paymentMethod);
   550→    setHeldOrdersOpen(false);
   551→    toast.success(t("orderResumed"), { duration: 1500 });
   552→  };
   553→
   554→  const handleDeleteHeldOrder = (id: string) => {
   555→    deleteHeldOrder(id);
   556→    toast.success(t("heldOrderDiscarded"));
   557→  };
   558→
   559→  // Keyboard shortcuts: F1 focus scan, F2/F3/F4 payment methods, F6 hold, F7 held, F9 checkout
   560→  useEffect(() => {
   561→    const handler = (e: KeyboardEvent) => {
   562→      // Skip if a dialog is open or user is typing in a non-scan input
   563→      const target = e.target as HTMLElement | null;
   564→      const isTyping =
   565→        target?.tagName === "INPUT" ||
   566→        target?.tagName === "TEXTAREA" ||
   567→        target?.isContentEditable;
   568→
   569→      // F1 — focus scan input (always works, even if typing)
   570→      if (e.key === "F1") {
   571→        e.preventDefault();
   572→        scanInputRef.current?.focus();
   573→        scanInputRef.current?.select();
   574→        return;
   575→      }
   576→
   577→      // F2/F3/F4 — payment method selection (only when cart non-empty)
   578→      if (e.key === "F2" && items.length > 0) {
   579→        e.preventDefault();
   580→        setPaymentMethod("Cash");
   581→        toast.success(`${t("paymentMethod")}: ${t("cash")}`, { duration: 1000 });
   582→        return;
   583→      }
   584→      if (e.key === "F3" && items.length > 0) {
   585→        e.preventDefault();
   586→        setPaymentMethod("UPI");
   587→        toast.success(`${t("paymentMethod")}: ${t("upi")}`, { duration: 1000 });
   588→        return;
